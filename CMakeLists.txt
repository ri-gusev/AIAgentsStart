cmake_minimum_required(VERSION 3.15)
project(openai_cli LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_program(RAG_PYTHON NAMES python.exe python3 python
    PATHS "${CMAKE_CURRENT_SOURCE_DIR}/.venv-rag/bin" "${CMAKE_CURRENT_SOURCE_DIR}/.venv-rag/Scripts" NO_DEFAULT_PATH)
find_program(RAG_PYTHON NAMES python.exe python3 python)
if(NOT RAG_PYTHON)
    message(FATAL_ERROR "Python is required for RAG document extraction")
endif()
add_compile_definitions(RAG_PYTHON="${RAG_PYTHON}" RAG_EXTRACTOR="${CMAKE_CURRENT_SOURCE_DIR}/rag_extract.py")
find_program(RAG_PYTHON NAMES python.exe python3 python
    PATHS "${CMAKE_CURRENT_SOURCE_DIR}/.venv-rag/bin" "${CMAKE_CURRENT_SOURCE_DIR}/.venv-rag/Scripts" NO_DEFAULT_PATH)
find_program(RAG_PYTHON NAMES python.exe python3 python)
if(NOT RAG_PYTHON)
    message(FATAL_ERROR "Python is required for RAG document extraction")
endif()
add_compile_definitions(RAG_PYTHON="${RAG_PYTHON}" RAG_EXTRACTOR="${CMAKE_CURRENT_SOURCE_DIR}/rag_extract.py")

# CMake 4 exposes the modern SQLite3::SQLite3 target, while older supported
# CMake releases used SQLite::SQLite3.
if(TARGET SQLite3::SQLite3)
    set(SQLITE_TARGET SQLite3::SQLite3)
else()
    set(SQLITE_TARGET SQLite::SQLite3)
endif()

set(REMINDER_SOURCES reminder_store.cpp reminder_scheduler.cpp reminder_events.cpp reminder_pipeline.cpp)
set(MCP_SOURCES mcp_client.cpp mcp_manager.cpp codeforces_watcher.cpp)
add_executable(openai_cli main.cpp agent.cpp api_client.cpp web_server.cpp document_indexer.cpp rag_retrieval.cpp memory_store.cpp invariant_store.cpp ${MCP_SOURCES} ${REMINDER_SOURCES})
target_link_libraries(openai_cli PRIVATE CURL::libcurl ${SQLITE_TARGET} Threads::Threads)

if(WIN32)
    target_link_libraries(openai_cli PRIVATE ws2_32 bcrypt crypt32)
endif()
